

set(SERVICE_LIB_SOURCE_DIR src)

set(SERVICE_LIB_SOURCE_CODES
    ${SERVICE_LIB_SOURCE_DIR}/service_pipeline.cpp
)

add_library(${SERVICE_LIB_NAME} STATIC ${SERVICE_LIB_SOURCE_CODES})
target_compile_definitions(${SERVICE_LIB_NAME}
    PUBLIC SERVICE_NAMESPACE_PREFIX=${SERVICE_NAMESPACE_PREFIX}
    PUBLIC DEBUG_ENV_FILE="${DEBUG_ENV_FILE}"
    PUBLIC ENV_VARIABLES_PREFIX="${ENV_VARIABLES_PREFIX}")

target_include_directories(${SERVICE_LIB_NAME}
    PRIVATE ${CONFIGS_INCLUDE_DIR}
    )

target_link_libraries(${SERVICE_LIB_NAME}
    PRIVATE ${ZMQRPC_PACKAGE}
    PRIVATE FRZGatewayLib
    PRIVATE system::ecal
    PRIVATE conan::msgpack
    PRIVATE conan::mgsl
    PRIVATE conan::libenvpp
    PRIVATE conan::CLI11
)

set_target_properties(
    ${SERVICE_LIB_NAME}
    PROPERTIES
    ARCHIVE_OUTPUT_DIRECTORY_DEBUG ${CMAKE_BINARY_DIR}/lib/Debug
    ARCHIVE_OUTPUT_DIRECTORY_RELEASE ${CMAKE_BINARY_DIR}/lib/Release
    LIBRARY_OUTPUT_DIRECTORY_DEBUG ${CMAKE_BINARY_DIR}/lib/Debug
    LIBRARY_OUTPUT_DIRECTORY_RELEASE ${CMAKE_BINARY_DIR}/lib/Release
)


install(TARGETS ${SERVICE_LIB_NAME}
    CONFIGURATIONS Debug Release
    ARCHIVE DESTINATION lib
    LIBRARY DESTINATION bin
)
